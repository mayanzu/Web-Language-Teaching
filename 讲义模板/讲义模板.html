一、章节标题（使用中文数字）

这是普通段落文本。段落是讲义的基础组成部分，用于解释概念、描述知识点。段落之间会有适当的间距，保持良好的阅读体验。

第二个段落示例。可以在段落中使用粗体来强调重要内容，使用行内代码来表示代码片段、属性名或关键字。

5.4 带注释的代码块

// 这是一个带有详细注释的代码示例
let score = 0;           // 初始化分数
let playerName = '玩家'; // 玩家名称

// 增加分数的函数
function addScore(points) {
    score += points;     // 累加分数
    console.log(`${playerName} 获得 ${points} 分，当前总分：${score}`);
}

addScore(100); // 输出: 玩家 获得 100 分，当前总分：100
addScore(50);  // 输出: 玩家 获得 50 分，当前总分：150
